if (NOT DEFINED ENV{SYSROOT})
	message(FATAL_ERROR "SYSROOT must be defined")
endif()

set(ETHEREAL_SYSROOT $ENV{SYSROOT})

set(CMAKE_SYSTEM_NAME Ethereal)
set(CMAKE_SYSTEM_PROCESSOR x86_64)

set(CMAKE_SYSROOT $ENV{SYSROOT})
set(CMAKE_STAGING_PREFIX $ENV{SYSROOT}/usr)

set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_PACKAGE BOTH)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED True)
set(CMAKE_CXX_COMPILER x86_64-ethereal-g++)
set(CMAKE_CXX_COMPILER_WORKS True)

set(CMAKE_C_COMPILER x86_64-ethereal-gcc)
set(CMAKE_C_COMPILER_WORKS True)
